
datasource db {
  provider = "postgresql"
  url      = env("DATABASE_URL")
}

generator client {
  provider = "prisma-client-js"
}

// Bảng Bệnh nhân (Giữ nguyên để tham khảo nếu cần)
model Patient {
  id            String   @id @default(uuid())
  recordNumber  String   @unique
  name          String
  dob           DateTime
  gender        String
  phone         String?
  address       String?
  identityCard  String?
  insuranceCard String?
  
  createdAt     DateTime @default(now())
  updatedAt     DateTime @updatedAt
}

// Bảng dữ liệu gửi giấy tờ (Theo cấu trúc người dùng cung cấp)
model BhDocument {
  id              Int       @id @default(autoincrement())
  patientId       String?   @db.VarChar(20)
  patientName     String?   @map("patientname") @db.VarChar(80)
  docNo           String?   @map("docno") @db.VarChar(20)
  typeCode        String?   @map("typeCode") @db.VarChar(10)
  typeName        String?   @map("typeName") @db.VarChar(60)
  xmlData         String?   @map("xmlData") @db.VarChar(4080)
  signatureStatus String?   @map("signatureStatus") @db.VarChar(20)
  sendStatus      String?   @map("sendStatus") @db.VarChar(20)
  sentAt          DateTime? @default(now()) @map("sentAt")
  transactionId   String?   @map("transactionId") @db.VarChar(20)
  errorMessage    String?   @map("errorMessage") @db.VarChar(20) // Lưu ý: 20 ký tự khá ngắn cho thông báo lỗi
  createdAt       DateTime? @default(now()) @map("createdAt")
  updatedAt       DateTime? @updatedAt @map("updatedAt")

  @@map("bh_document")
}
